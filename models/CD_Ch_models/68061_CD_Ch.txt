' Enumerations
enum Department {
  SALES
  MARKETING
  OTHER
}

enum SweetnessLevel {
  DRY
  SEMI_DRY
  SEMI_SWEET
  SWEET
}

enum PriorityLevel {
  LEVEL_1
  LEVEL_2
  LEVEL_3
  LEVEL_4
  LEVEL_5
}

enum OrderStatus {
  IN_PROCESS
  DISPATCHED
  DELIVERED
  PAID
  SUSPENDED
  CANCELED
}

enum OrderType {
  REGULAR
  URGENT
}


class Order {
  + orderNumber: int
  + orderDate: Date
  + status: OrderStatus
  + shipmentDate: Date
  + assignedEmployeeID: int

}

  + priorityLevel: PriorityLevel
  + expectedDeliveryTime: String

}

class RegularOrder extends Order {
  + mainCustomerID: int
  + consolidateJointOrder(): void
  + minimizeDeliveryCosts(): void
}

  + employeeID: int

  + viewWineCatalog(): void
