' People Classes
'==========================

abstract class Person {
  -ID : int
  -firstName : String
  -lastName : String
  -phoneNumber : String
  -emailAddress : String
  +viewCatalog(): void
}

class Customer {
  -deliveryStreetAddress : String
  -deliveryCity : String
  -deliveryState : String
  -deliveryZipCode : String
  -dateOfContact : Date
  +registerAsCustomer(): void
  +viewOrderStatus(order: Order): String
}

class Manufacturer {
  -manufacturerID : int
  -manufacturerFirstName : String
  -manufacturerLastName : String
  -manufacturerPhoneNumber : String
  -manufacturerEmail : String
  -manufacturerStreetAddress : String
  -manufacturerCity : String
  -manufacturerState : String
  -manufacturerZipCode : String
}

abstract class Employee {
  -officeStreetAddress : String
  -officeCity : String
  -officeState : String
  -officeZipCode : String
  -workStartDate : Date
  +viewOrders(): void
}

class SalesEmployee {
  +addOrder(order:Order): void
  +updateInventory(wineId:int, quantity:int, storageID:int): void
  +updateOrderStatus(order:Order, status:CurrentStatus): void
  +generateUnproductiveEmployeeReport(startDate:Date, endDate:Date): int
  +generateWineRecommendationReport(dishName:String, occasion:String, wineType:String): List<Wine>
  +joinCustomerToOrder(customerID:int, orderId:int, newOrderID:int): void
  +checkInventory(wineID:int): int

}

class MarketingEmployee {
  +updateWinePrice(wine:Wine, price:double) : void
  +updateWinePicture(wine:Wine, image:String): void
  +addFoodPairingss(wineType:WineType, pairing:FoodPairings): void
  +updateWineRecommendation(wineTypeID:int, season:enum, occasion:String, drinkInDoor:boolean): void
}

(Customer, Employee) --|> Person : {complete, disjoint}
(SalesEmployee, MarketingEmployee) --|> Employee : {complete, disjoint}

class Order {
  -orderID : int
  -orderDate : Date
  -currentStatus : CurrentStatus
  -shipmentDate : Date
  -employeeID : int
  -urgent : boolean
}


class UrgentOrder {
  -urgentOrderID : int
  -priorityLevel : priorityLevel
  -expectedDeliveryTime : DateTime
}

class CustomerWineSelection {
  -customerID : int
  -orderID : int
  -wineID : int
  -quantity : int
}

class JointCustomerOrder {
  -customerID : int
  -orderID : int
  -isMainCustomer : boolean
}

Order <|-- UrgentOrder : {complete, disjoint}

' Relationships for orders
Order "1" <--* "1..*" CustomerWineSelection : consistsOf
Customer "1" <--* "1..*" CustomerWineSelection : places
JointCustomerOrder "0..*" -- "1..*" Order : includes
JointCustomerOrder "0..*" -- "1" Customer : participates
Order ..> CurrentStatus : hasStatus >
UrgentOrder ..> PriorityLevel : hasPriority >
