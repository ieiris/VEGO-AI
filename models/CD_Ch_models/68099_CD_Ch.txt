@startuml
class Customer {
  - id: int
  - name: String
  - address: Address
  - phoneNumber: PhoneNumber
  - email: EmailAddress
  - firstContactDate: Date
  + trackOrderStatus(order: Order)
  + viewWineInformation(wine: Wine)
}

abstract class Order {
  - id: int
  - orderNumber: String
  - orderDate: Date
  - status: OrderStatus
  - shipmentDate: Date
  - assignedEmployee: Employee
  + getWines(): List<Wine>
  + getCustomers(): List<Customer>
}

enum OrderStatus {
  IN_PROCESS
  DISPATCHED
  DELIVERED
  PAID
  SUSPENDED
  CANCELED
}

Order --* OrderStatus

class UrgentOrder {
  - priorityLevel: int
  + getPriorityLevel(): int
}

class RegularOrder {
  - expectedDeliveryDate: Date
  + getExpectedDeliveryDate(): Date
}

UrgentOrder --|> Order
RegularOrder --|> Order

abstract class Employee {
  - id: int
  - name: String
  - phoneNumber: PhoneNumber
  - officeAddress: Address
  - email: EmailAddress
  - employmentStartDate: Date
  + updateInformation(information: EmployeeInformation)
  + getOrderHistory(): List<Order>
}

class SalesStaff {
  + processOrders(orders: List<Order>)
  + generateSalesReport(): String
}

class MarketingStaff {
  + updateWineInformation(wine: Wine)
  + generateMarketingReport(): String
}

SalesStaff --|> Employee
MarketingStaff --|> Employee

class Wine {
  - id: int
  - name: String
  - serialNumber: String
  - type: WineType
  - vintage: String
  - region: String
  - grapeVariety: String
  - foodPairings: List<String>
  - occasionRecommendations: List<String>
  - price: double
  - quantity: int
  - sweetnessLevel: SweetnessLevel
  + getWineInformation(): String
}

enum WineType {
  RED
  WHITE
  SPARKLING
  FORTIFIED
  ROSÉ
  DESSERT
}

Wine --* WineType

enum SweetnessLevel {
  DRY
  SEMI_DRY
  SEMI_SWEET
  SWEET
}

Wine --* SweetnessLevel

class Manufacturer {
  - id: int
  - name: String
  - contactPhoneNumber: PhoneNumber
  - address: Address
  - email: EmailAddress
  + getWines(): List<Wine>
}

class StorageLocation {
  - id: int
  - location: Location
  + getWines(): List<Wine>
}

class Gefen {
  + provideWineProducerInformation(): String
  + provideWineInformation(): String
}
