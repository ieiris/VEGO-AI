' Priority Level Enum for UrgentOrder
enum PriorityLevel {
  LOW = 1
  MEDIUM = 2
  HIGH = 3
  VERY_HIGH = 4
  URGENT = 5
}

enum Location {
  INDOOR
  OUTDOOR
}

' UrgentOrder Class now includes PriorityLevel enum
class UrgentOrder {
  + priorityLevel: PriorityLevel

}
