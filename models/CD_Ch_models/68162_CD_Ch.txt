abstract class Person {
  - personID: Long
 
  
}

abstract class Employee extends Person {
  - name: String
  - phoneNumber: String
  - email: String
  - officeAddress: String
  - employmentStartDate: Date
  + placeOrder()
  + viewOrderHistory()
  + updateEmployeeInfo()
  + browseWine()
}

class Customer extends Person {
  - name: String
  - phoneNumber: String
  - email: String
  - deliveryAddress: String
  - dateOfFirstContact: Date
  - foodPreferences: List<String>
  - occasionPreferences: List<String>
  - wineTypePreferences: List<String>
  + placeOrder()
  + viewOrderHistory()
  + updateCustomerInfo()
  + getPurchaseHistory()
  + setPreferences(foodPreferences: List<String>, occasionPreferences: List<String>, wineTypePreferences: List<String>)
  + getPreferences(): Map<String, Object>
}

class SalesEmployee extends Employee {
  + generateRecommendations()
  + generateUnproductiveEmployeeReport(Date date )
  + assignOrder()
}

class MarketingEmployee extends Employee {
  + updateWineInfo()
  + addOccasionRecommendation()
}

enum OrderStatus {
  IN_PROCESS
  DISPATCHED
  DELIVERED
  PAID
  SUSPENDED
  CANCELED
}

enum Sweetness {
  DRY
  SEMI_DRY
  SEMI_SWEET
  SWEET
}

enum LocationType {
  INDOOR
  OUTDOOR
}

enum Season {
  WINTER
  SUMMER
  SPRING
  AUTUMN
}

abstract class Order {
  - orderID: int
  - orderDate: Date
  - status: OrderStatus
  - shipmentDate: Date
  - assignedEmployee: Employee
  + createOrderAndUpdateInventory()
  + updateOrderAndInventoryStatus()
  + cancelOrder()
  + calculateTotalQuantity()
}

class RegularOrder extends Order {
  - mainCustomer: Customer
  - orderItems: HashMap<Wine, Integer>
  + addWineToOrder(Wine wine, int quantity)
  + removeWineFromOrder(Wine wine, int quantity)
}

class UrgentOrder extends RegularOrder {
  - priorityLevel: int
  - expectedDeliveryTime: Date
}

class JointOrder extends Order {
  - mainCustomer: Customer
  - customersAndOrders: HashMap<Customer, HashMap<Wine, Integer>>
  + addCustomersAndWineToOrder(HashMap<Customer, HashMap<Wine, Integer>>)
  + removeCustomersFromOrder(List<Customer> customers)
  + addWineToOrder(Wine wine, int quantity)
  + removeWineFromOrder(Wine wine, int quantity)
}

class WineType {
  - type: String
  - serialNumber: double
  - foodPairings: List<String>
  - occasionRecommendations: List<String>
  - recipeLinks: List<String>
  - wines: List<Wine>
  + addFoodPairing(String foodPairing)
  + removeFoodPairing(String foodPairing)
  + addOccasionRecommendation(String occasionRecommendation)
  + removeOccasionRecommendation(String occasionRecommendation)
  + getFoodPairings()
  + getOccasionRecommendations()
}

class Occasion {
  - description: String
  - seasons: List<Season>
  - locationTypes: List<LocationType>
}

class Wine {
  - manufacturerNumber: double
  - catalogNumber: double
  - name: String
  - picture: String
  - pricePerBottle: double
  - storageLocations: List<StorageLocation>
  - sweetness: Sweetness
  - Description: String
  - Production Year: int
}

class StorageLocation {
  - storageLocationID: int
  - name: String
  - uniqueNumber: int
  - locationType: LocationType
  - inventory: HashMap<Wine, Integer>
  + addInventory(Wine wine, int quantity)
  + removeInventory(Wine wine, int quantity)
}

class Manufacturer {
  - manufacturerID: int
  - fullName: String
  - contactPhoneNumber: String
  - address: String
  - email: String
}
