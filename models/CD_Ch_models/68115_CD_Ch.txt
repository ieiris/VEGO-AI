!define CLASS_COLOR #66CCCC
!define ABSTRACT_CLASS_COLOR #99CCFF

top to bottom direction

class "Employee" as employee << (C,CLASS_COLOR) >>
{
  - id: Integer
  - name: String
  - phoneNumber: String
  - officeAddress: String
  - email: String
  - employmentStartDate: Date
}

class "SalesEmployee" as salesEmployee << (C,CLASS_COLOR) >>
{
  - employeeId: Integer
  - numberOfUrgentOrders: Integer
  - numberOfRegularOrders: Integer
  + updateStock(order: Order): void
}
salesEmployee --|> employee

class "MarketingEmployee" as marketingEmployee << (C,CLASS_COLOR) >>
{
  - employeeId: Integer
}
marketingEmployee --|> employee

class "Admin" as admin << (C,CLASS_COLOR) >>
{
  - employeeId: Integer
  + manageUsers(user: Employee): void
  + assignPermissions(user: Employee, role: String): void
}
admin --|> employee

class "Customer" as customer << (C,CLASS_COLOR) >>
{
  - id: Integer
  - name: String
  - phoneNumber: String
  - deliveryAddress: String
  - email: String
  - dateOfFirstContact: Date
}

enum OrderStatus {
  IN_PROCESS
  DISPATCHED
  DELIVERED
  PAID
  SUSPENDED
  CANCELED
}

class "Order" as order << (C,CLASS_COLOR) >>
{
  - id: Integer
  - orderDate: Date
  - status: OrderStatus
  - shipmentDate: Date
  - assignedEmployee: SalesEmployee
  - customer: Customer
}
order --* salesEmployee
order --* customer
order *-- orderDetail : contains
order --* OrderStatus

class "OrderDetail" as orderDetail << (C,CLASS_COLOR) >>
{
  - wine: Wine
  - quantity: Integer
}
orderDetail --* wine

class "UrgentOrder" as urgentOrder << (C,CLASS_COLOR) >>
{
  - priorityLevel: Integer
  - expectedDeliveryTime: Date
}
urgentOrder --|> order

class "RegularOrder" as regularOrder << (C,CLASS_COLOR) >>
{
  - jointOrder: Boolean
  - customers: List<Customer>
}
regularOrder --|> order

class "Wine" as wine << (C,CLASS_COLOR) >>
